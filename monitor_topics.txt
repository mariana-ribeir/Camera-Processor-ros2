# Comandos para monitorear los tópicos de pose en ROS2
# Ejecuta cada uno en una terminal separada dentro del contenedor
# Agrupados por nodos que publican los tópicos

# === CAMERA_SIMULATOR ===
# 1. Monitorear /camera/image_raw (imagen cruda de la cámara)
docker exec -it camera_processor_ws bash -c 'source /opt/ros/jazzy/setup.bash && source /workspaces/ros2_ws/install/setup.bash && ros2 topic echo /camera/image_raw'

# === PERSON_PROCESSOR ===
# 2. Monitorear /person/detected (detecciones de personas)
